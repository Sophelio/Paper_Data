    def derivative(self, x, dt=1):

        def do_deriv(some_x):
            dx       = np.zeros_like(some_x, dtype=np.float64)
            # central difference  
            dx[1:-1] = 0.5 * (some_x[2:] - some_x[:-2]) / dt
            # boundary conditions
            dx[0]    = (some_x[1] - some_x[0]) / dt
            dx[-1]   = (some_x[-1] - some_x[-2]) / dt
