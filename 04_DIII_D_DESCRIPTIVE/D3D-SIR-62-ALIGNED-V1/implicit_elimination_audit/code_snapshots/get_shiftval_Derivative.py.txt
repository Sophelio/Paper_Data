def get_shiftval(x, dx, d2x, d3x, d4x, nmin, out_type):
    
    if out_type == 'Variable':
        blob = np.vstack([x])
        shiftval = abs(blob.min()) 
        shiftval += nmin
        del blob

    elif out_type == 'Derivative':
        blob = np.vstack([x,dx])
        shiftval = abs(blob.min()) 
        shiftval += nmin
        del blob
