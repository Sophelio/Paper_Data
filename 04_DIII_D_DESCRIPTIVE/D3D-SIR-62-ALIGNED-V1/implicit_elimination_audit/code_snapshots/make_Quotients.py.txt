    def make_Quotients(self, VAR, VAR_keys, shift, add_inverse = False):
        
        X, X_keys = VAR, VAR_keys

        quotients, q_names = [], [] 
        idx                = np.arange(len(X))
        idx_combs          = list(combinations(idx,2))
        
        for c in idx_combs:
            # quotient
            name_token = X_keys[c[0]]/X_keys[c[1]]
            q_names.append(name_token)
            i0,i1      = c
            out        = (X[i0]+shift) / (X[i1]+shift)
            quotients.append(out)

            if add_inverse:
                # inverse of quotient
                name_token_inv = X_keys[c[1]]/X_keys[c[0]]
                q_names.append(name_token_inv)
                quotients.append(np.power(out,-1))
        
        return np.array(quotients), q_names
